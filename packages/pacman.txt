# ── Package tooling ───────────────────────────────────────────────────
# NOT on a stock CachyOS install — verified in a VM. The Flatpak step below
# depends on this, so it has to be installed here first. (CachyOS ships
# `shelly`, which can front for Flatpak, but it delegates to this same
# binary — without it, shelly can't see Flathub either.)
flatpak

# ── Build toolchain (needed by AgentTileCLI + StreamHub) ──────────────
git
github-cli                   # the `gh` command

rust                         # cargo — builds AgentTileCLI
pkgconf
# The next four are AgentTileCLI's build deps, exactly what its own
# install.sh preflight checks for (and its README's Arch hint installs).
# libadwaita happens to ride in as a dependency of lact today, but that's
# lact's business — name everything the build needs, not just what's left.
gtk4                         # AgentTileCLI: GTK4 >= 4.12
vte4                         # AgentTileCLI: terminal widget, VTE >= 0.70
libadwaita                   # AgentTileCLI: app chrome (header bar, dialogs), >= 1.5
gtksourceview5               # AgentTileCLI: the sidebar's file editor
nodejs                       # StreamHub (only if building from source)
npm
fuse2                        # StreamHub + ConsoleVault: AppImages need libfuse.so.2
minisign                     # verifies ConsoleVault's + Music AI Player's signed AppImages

# ── Music AI Player ───────────────────────────────────────────────────
# The player itself bundles everything it needs; these two are for its YouTube
# importer, which shells out to whatever is on PATH. Without them the app still
# starts and plays — the Import panel just refuses with "yt-dlp is not
# installed". ffmpeg is what yt-dlp extracts audio with, and ffprobe (same
# package) is what measures each track's real duration for the crossfade.
yt-dlp                       # Music AI Player: YouTube import
ffmpeg                       # Music AI Player: yt-dlp's audio extraction + ffprobe

# ── KDE rice (kderice) ────────────────────────────────────────────────
# Wallpapers are generated, not shipped: 51 terrain variants rendered at
# every output resolution.
python-numpy                 # kderice: wallpaper generation
python-pillow                # kderice: wallpaper generation
# The slides are lossless WebP. Qt reads WebP only through this plugin —
# without it Plasma shows a black desktop and no error.
qt6-imageformats
# 'kderice apply' hard-fails without this font. Installing it here also
# means 'kderice setup' never has to sudo for it, so it can't stop to
# prompt in the middle of an unattended run.
ttf-fira-mono
kscreen                      # kscreen-doctor — the monitor-geometry gate
rsync                        # kderice apply installs the wallpapers with it
